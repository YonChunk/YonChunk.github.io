for(auto it = v.begin(); it != v.end(); ++it) {
mx = std::max(*it, mx);
}
int ans = 0;
int lt = 0, rt = mx;
while(lt <= rt) {
int mid = lt + (rt - lt) / 2;
if (sum(v, mid) == obj) {
return mid;
}
else if(sum(v, mid) > obj) {
ans = mid;
lt = mid + 1;
}
else rt = mid - 1;
}
return ans;
}
